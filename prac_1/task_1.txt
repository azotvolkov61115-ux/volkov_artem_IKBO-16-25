    first_line=$(head -n 1 "$file")

     extension="${file##*.}"

     if [[ "$extension" == "py" && "$first_line" =~ ^[[:space:]]*# ]] ||
         [[ "$extension" != "py" && "$first_line" =~ ^[[:space:]]*(//|/\*) ]]; then
        echo "$file: первая строка содержит комментарий"
    else

    fi
done

задание 8:
    #!/bin/bash

    if [[ $# -ne 2 ]]; then
        printf 'Использование: %s КАТАЛОГ РАСШИРЕНИЕ\n' "$0" >&2
        exit 1
    fi

    directory=$1
    extension=${2#.}

    if [[ ! -d "$directory" || -z "$extension" || "$extension" == */* ]]; then
        printf 'Укажите существующий каталог и корректное расширение.\n' >&2
        exit 1
    fi

    archive="${extension}_files.tar"
    mapfile -d '' files < <(
        cd -- "$directory" &&
        find . -type f -name "*.${extension}" ! -path "./${archive}" -print0
    )

    if [[ ${#files[@]} -eq 0 ]]; then
        printf 'Файлы с расширением .%s не найдены.\n' "$extension"
        exit 0
    fi

    tar -cf "$archive" -C "$directory" -- "${files[@]}"

    #!/bin/bash

    if [[ $# -ne 2 ]]; then
        printf 'Использование: %s ВХОДНОЙ_ФАЙЛ ВЫХОДНОЙ_ФАЙЛ\n' "$0" >&2
        exit 1
    fi

    input=$1
    output=$2

    if [[ ! -r "$input" ]]; then
        printf 'Входной файл недоступен для чтения: %s\n' "$input" >&2
        exit 1
    fi

    if [[ "$input" == "$output" || ( -e "$output" && "$input" -ef "$output" ) ]]; then
        printf 'Входной и выходной файлы должны быть разными.\n' >&2
        exit 1
    fi

    sed 's/    /\t/g' -- "$input" > "$output"
